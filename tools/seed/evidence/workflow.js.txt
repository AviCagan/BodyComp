const batches = args.batches

phase('Research')
log('Researching ' + batches.length + ' movement areas against PubMed and primary sources')
const results = await pipeline(
  batches,
  (b) => agent(
    [
      'You are building the evidence base for a workout app\'s exercise-to-muscle credit table. Today is 2026-09-08.',
      TAXONOMY, '', RULES, '',
      'BATCH: ' + b.batch + ' (' + b.count + ' exercises)',
      'SCOPE: ' + b.scope,
      'BATCH FILE (read it with the Read tool): ' + b.file,
      'Each exercise entry has: id, name, equipment, mechanic, pattern, and `current` = the mapping the app uses today (g = groupId, w = weight, d = regionDistribution, flag = the rule was a guess). Improve on `current` only where evidence or anatomy says so; where the current mapping is already right, keep it and say why in one line.',
      'Also read /home/user/BodyComp/src/engine/taxonomy.ts if you need the exact ids.',
    ].join('\n'),
    { label: 'research:' + b.batch, phase: 'Research', schema: FAMILY_RESULT, effort: 'high' },
  ).then(r => r ? Object.assign({ file: b.file, scope: b.scope }, r) : null),
  (research, b) => {
    if (!research) return null
    return agent(
      [
        'You are an independent, skeptical verifier for an exercise-to-muscle evidence table. Today is 2026-09-08.',
        TAXONOMY, '', RULES, '',
        'A researcher produced the family mappings below for the batch "' + b.batch + '" (' + b.scope + '). The batch file with every exercise is at ' + b.file + ' (read it).',
        'Check, for EVERY family:',
        '1. Citations: look each one up (PubMed MCP tools via ToolSearch "select:mcp__PubMed__search_articles,mcp__PubMed__get_article_metadata,mcp__PubMed__lookup_article_by_citation", or WebSearch for preprints/books). Does it exist? Does it actually support the claim it is attached to? Mark hallucinated or misattributed citations.',
        '2. Rule compliance: at least one 1.0 weight; weights within 0.25..1.0; a distribution present for every multi-region group credited, keys valid for that group, summing to 1.0 (tolerance 0.001); only valid groupIds; anchor rows untouched.',
        '3. Assignment sanity: spot-check at least 15 exercise ids against their names; flag any exercise placed in a family that does not fit (e.g. a decline press in the incline family, a seated calf raise in the standing family). Flag ids from the batch that appear in no family and were not listed as unassigned, and ids listed twice.',
        '4. Substance: does the mapping follow from the evidence tier claimed? Is a strong claim resting only on consensus? Are secondary credits inflated (everything at 0.5) or missing (an obvious synergist omitted)?',
        'Verdict per family: accept (use as is), revise (use with the listed corrections), reject (send back). Be concrete: every correction names the field, the current value and the value it should be. Do not modify any repository file. Return ONLY the structured output.',
        '',
        'RESEARCHER OUTPUT:',
        JSON.stringify(research, null, 1).slice(0, 150000),
      ].join('\n'),
      { label: 'verify:' + b.batch, phase: 'Verify', schema: VERIFY_RESULT, effort: 'high' },
    ).then(v => ({ batch: b.batch, research, verification: v }))
  },
)

const done = results.filter(Boolean)
log('Completed ' + done.length + '/' + batches.length + ' batches')
if (done.length < batches.length) log('Missing: ' + batches.filter(b => !done.find(d => d.batch === b.batch)).map(b => b.batch).join(', '))
return done
